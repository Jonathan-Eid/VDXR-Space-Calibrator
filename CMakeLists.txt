cmake_minimum_required(VERSION 3.21)
project(VDXRSpaceCalibrator VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The current renderer and runtime integration target Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
set(SURVIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/external/libsurvive" CACHE PATH "Pinned libsurvive source")
set(SURVIVE_LIBRARY "" CACHE FILEPATH "Double-precision libsurvive import library")
set(EIGEN_INCLUDE_DIR "" CACHE PATH "Eigen headers from the libsurvive build")
if(NOT SURVIVE_LIBRARY OR NOT EIGEN_INCLUDE_DIR)
  message(FATAL_ERROR "Use scripts/Build.ps1 or supply SURVIVE_LIBRARY and EIGEN_INCLUDE_DIR")
endif()
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/imgui")
add_library(calibration-imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp"
  "${IMGUI_DIR}/imgui_tables.cpp" "${IMGUI_DIR}/imgui_widgets.cpp" "${IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
target_include_directories(calibration-imgui PUBLIC "${IMGUI_DIR}")
add_executable(vdxr-calibration WIN32 src/calibration_app.cpp external/cJSON/cJSON.c)
target_include_directories(vdxr-calibration PRIVATE include "${EIGEN_INCLUDE_DIR}" external/OpenXR-SDK/include external/cJSON
  "${SURVIVE_SOURCE}/include" "${SURVIVE_SOURCE}/redist" "${SURVIVE_SOURCE}/libs/cnmatrix/include")
target_compile_definitions(vdxr-calibration PRIVATE USE_DOUBLE SURVIVE_EXPORT=__declspec\(dllimport\))
target_compile_options(vdxr-calibration PRIVATE /bigobj)
target_link_libraries(vdxr-calibration PRIVATE calibration-imgui "${SURVIVE_LIBRARY}" d3d11 dxgi d3dcompiler shell32 windowscodecs ole32)
add_executable(calibration-math-test src/calibration_math_test.cpp)
target_include_directories(calibration-math-test PRIVATE "${EIGEN_INCLUDE_DIR}")
enable_testing()
add_test(NAME calibration-math COMMAND calibration-math-test)
add_executable(device-identity-test src/device_identity_test.cpp external/cJSON/cJSON.c)
target_include_directories(device-identity-test PRIVATE "${EIGEN_INCLUDE_DIR}" external/cJSON)
add_test(NAME device-identity COMMAND device-identity-test)
add_executable(calibration-replay src/calibration_replay.cpp external/cJSON/cJSON.c)
target_include_directories(calibration-replay PRIVATE "${EIGEN_INCLUDE_DIR}" external/cJSON)
add_executable(survive-import-test src/survive_import_test.cpp)
target_include_directories(survive-import-test PRIVATE "${SURVIVE_SOURCE}/include" "${SURVIVE_SOURCE}/redist" "${SURVIVE_SOURCE}/libs/cnmatrix/include")
target_compile_definitions(survive-import-test PRIVATE USE_DOUBLE SURVIVE_EXPORT=__declspec\(dllimport\))
target_link_libraries(survive-import-test PRIVATE "${SURVIVE_LIBRARY}")
get_filename_component(SURVIVE_IMPORT_DIRECTORY "${SURVIVE_LIBRARY}" DIRECTORY)
get_filename_component(SURVIVE_BINARY_DIRECTORY "${SURVIVE_IMPORT_DIRECTORY}/../../Release" ABSOLUTE)
add_test(NAME survive-windows-import COMMAND survive-import-test)
set_tests_properties(survive-windows-import PROPERTIES WORKING_DIRECTORY "${SURVIVE_BINARY_DIRECTORY}")
